cmake_minimum_required(VERSION 3.22)
project(fileform_heic_decode LANGUAGES CXX)
set(HEIC_ROOT "" CACHE PATH "Pinned evaluation dependency prefix")
if(NOT EXISTS "${HEIC_ROOT}/include/libheif/heif.h")
  message(FATAL_ERROR "Use the pinned HEIC dependency prefix")
endif()
add_executable(fileform-heic-decode main.cpp)
target_compile_features(fileform-heic-decode PRIVATE cxx_std_20)
target_include_directories(fileform-heic-decode PRIVATE "${HEIC_ROOT}/include")
if(WIN32)
  find_library(HEIF_LIBRARY NAMES heif libheif PATHS "${HEIC_ROOT}/lib" NO_DEFAULT_PATH REQUIRED)
  target_compile_definitions(fileform-heic-decode PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_compile_options(fileform-heic-decode PRIVATE /W4 /WX /utf-8)
elseif(APPLE)
  set(HEIF_LIBRARY "${HEIC_ROOT}/lib/libheif.dylib")
  target_compile_options(fileform-heic-decode PRIVATE -Wall -Wextra -Werror)
  set_target_properties(fileform-heic-decode PROPERTIES BUILD_RPATH "${HEIC_ROOT}/lib")
else()
  message(FATAL_ERROR "Only macOS and Windows are evaluated")
endif()
target_link_libraries(fileform-heic-decode PRIVATE "${HEIF_LIBRARY}")
if(WIN32)
  file(GLOB HEIC_DLLS "${HEIC_ROOT}/bin/*heif*.dll" "${HEIC_ROOT}/bin/*de265*.dll")
  foreach(DLL ${HEIC_DLLS})
    add_custom_command(TARGET fileform-heic-decode POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${DLL}" $<TARGET_FILE_DIR:fileform-heic-decode>)
  endforeach()
endif()
